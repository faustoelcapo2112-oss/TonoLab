function median(values) { const sorted = [...values].sort((a, b) => a - b); const mid = Math.floor(sorted.length / 2); return sorted.length % 2 ? sorted[mid] : (sorted[mid - 1] + sorted[mid]) / 2; } function sampleSkin(ctx, face) { const { x, y, width, height } = face; // Muestras de frente y mejillas; evita ojos, pelo y boca. const regions = [ [0.50, 0.32, 0.13, 0.09], [0.34, 0.53, 0.13, 0.12], [0.66, 0.53, 0.13, 0.12], [0.50, 0.62, 0.10, 0.07] ]; const pixels = []; for (const [cx, cy, rw, rh] of regions) { const sx = Math.max(0, Math.round(x + width * (cx - rw / 2))); const sy = Math.max(0, Math.round(y + height * (cy - rh / 2))); const sw = Math.max(1, Math.round(width * rw)); const sh = Math.max(1, Math.round(height * rh)); try { const data = ctx.getImageData(sx, sy, sw, sh).data; for (let i = 0; i < data.length; i += 4) { const r = data[i]; const g = data[i + 1]; const b = data[i + 2]; const max = Math.max(r, g, b); const min = Math.min(r, g, b); const brightness = (r + g + b) / 3; // Descarta sombras muy profundas, reflejos y colores extremos. if ( brightness < 35 || max > 248 || max - min < 8 || r < g * 0.88 || r < b * 0.88 ) continue; pixels.push([r, g, b]); } } catch { // Ignora una zona que no se pueda leer. } } if (pixels.length < 8) return null; // Mediana por canal: reduce la influencia de sombras y píxeles aislados. return [ median(pixels.map(p => p[0])), median(pixels.map(p => p[1])), median(pixels.map(p => p[2])) ]; }
